# 说明
将orthofinder+hybpiper从重测序数据中提取到的直系同源基因比对到基因组中，查看其分布位置

# 参考流程
[gmap](https://www.codeleading.com/article/41042092482/)

# 软件安装
```bash
mamba install bioconda::gmap
```

# 从hybpiper的结果中提取对应物种的直系同源基因

```bash
# 从hybpiper的结果中提取对应物种的序列
path=/data/xiongtao/project/Rosaceae/Rosaceae_cytonuclear/orthofinder/hybsuite_out_orthofinder_2/03-Orthology_inference/HRS/01-Original_HRS_sequences

for name in ${path}/*.FNA;do
  tt=$(basename $name)
  nn=$(grep -c Crataegus_hupehensis $name)
  if [[ $nn -eq 1 ]];then
      seqkit grep -p "Crataegus_hupehensis" $name > Malus_domestica_ortho_genes/${tt}
  fi
 done
```

## 执行比对
```bash
gmap -t 20 \ 
-D gmapdb \
-d crataegus_index \
-f gff3_gene \
--npaths=1 \ 
--min-identity=0.75 \ # 从0.80降低到0.75 
--min-trimmed-coverage=0.50 \ # 从0.60降低到0.50
crataegus_all_merged.fa > mapping_dir/mapping_crataegus_homo.gff3

```

## 从GMAP BED提取精准坐标
```python
#!/usr/bin/env python3
"""
parse_gmap_bed.py
解析GMAP输出的BED文件，提取每个ortholog基因的染色体坐标
usage: python3 parse_gmap_bed.py gmap_output.bed
"""

import pandas as pd
import sys

def parse_gmap_bed(bed_file, min_score=800, output_prefix="ortholog_positions"):
    """
    解析GMAP BED12格式输出
    
    GMAP BED12列：
    0: chr, 1: start(0-based), 2: end, 3: name(基因名),
    4: score(identity*1000), 5: strand,
    6: thickStart, 7: thickEnd, 8: itemRgb,
    9: blockCount(外显子数), 10: blockSizes, 11: blockStarts
    """
    
    records = []
    
    with open(bed_file) as f:
        for line in f:
            if line.startswith("#") or line.strip() == "":
                continue
            
            fields = line.strip().split("\t")
            if len(fields) < 6:
                continue
            
            chr_name   = fields[0]
            start      = int(fields[1])      # 0-based
            end        = int(fields[2])      # 0-based exclusive
            gene_id    = fields[3]
            score      = int(fields[4])      # identity × 1000
            strand     = fields[5]
            
            # identity过滤（score=1000表示100%相同）
            if score < min_score:
                continue
            
            gene_length = end - start
            identity    = score / 1000
            
            # 解析外显子信息（如果有BED12格式）
            exon_count = int(fields[9]) if len(fields) >= 10 else None
            
            records.append({
                "gene_id"    : gene_id,
                "chr"        : chr_name,
                "start_0based" : start,
                "end_0based"   : end,
                "start_1based" : start + 1,   # GFF3/人类可读坐标
                "end_1based"   : end,
                "strand"     : strand,
                "identity"   : identity,
                "gene_span"  : gene_length,
                "exon_count" : exon_count
            })
    
    df = pd.DataFrame(records)
    
    if df.empty:
        print("警告：没有通过过滤的比对结果，请降低 min_score 阈值")
        return df
    
    print(f"通过过滤的比对数: {len(df)}")
    print(f"唯一基因数:       {df['gene_id'].nunique()}")
    print(f"\n染色体分布（前10）:")
    print(df['chr'].value_counts().head(10).to_string())
    print(f"\nIdentity分布:")
    print(df['identity'].describe().round(3).to_string())
    
    # 保存完整结果（TSV）
    df.to_csv(f"{output_prefix}_full.tsv", sep="\t", index=False)
    
    # 保存标准BED格式（0-based，用于BEDTools）
    bed_out = df[["chr","start_0based","end_0based","gene_id","identity","strand"]]
    bed_out.to_csv(f"{output_prefix}.bed", sep="\t", index=False, header=False)
    
    print(f"\n输出文件:")
    print(f"  {output_prefix}_full.tsv  （完整信息）")
    print(f"  {output_prefix}.bed       （BEDTools用）")
    
    return df

if __name__ == "__main__":
    bed_file = sys.argv[1] if len(sys.argv) > 1 else "gmap_output.bed"
    parse_gmap_bed(bed_file, min_score=800)
```
