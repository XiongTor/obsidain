### 7.重复序列占比
**如何计算每个重复序列类型的占比？**

#1. 先算每个真实类别的总数（行和）

比如 LTR/Copia 总数 = 48704 + 554 + 2045 + ... + 37

#2. 你可以算每个真实类别占所有序列的比例

总数 = 所有行所有列（除 Misclas_rate 列）数值加起来

#3. 也可以计算每个真实类别正确分类的比例（准确率）

正确分类数 = 对角线数值

正确率 = 对角线 / 行和

#/data/xueqin/Prunus/genome_assembly/1Chromosome/06_polish/07.EDTA/T2T-Polish.fa.mod.EDTA.anno
#重复序列统计（总的：TE统计）
#1. 统计 TE 片段总长度（已去重叠）
# 取 TEanno.gff3 的 chr/start/end 列 ➜ 合并重叠 ➜ 求长度
grep -v '^#' T2T-Polish.fa.mod.EDTA.TEanno.gff3 \
  | awk '{print $1"\t"$4-1"\t"$5}' \
  | sort -k1,1 -k2,2n \
  | bedtools merge -i - \
  | awk '{sum += $3-$2} END{print "TE_total_bp\t"sum}' > TE_length.txt
#TE_total_bp     515636425

#2. 得到 基因组总长度
samtools faidx T2T-Polish.fa
awk '{sum+=$2} END{print "Genome_total_bp\t"sum}' T2T-Polish.fa.fai > genome_len.txt 
#Genome_total_bp 915815529

#3. 计算占比
paste TE_length.txt genome_len.txt \
  | awk '{printf("Repeat fraction = %.2f%%\n",$2/$4*100)}'
#56.303%

#/data/xueqin/Prunus/genome_assembly/1Chromosome/06_polish/07.EDTA/T2T-Polish.fa.mod.EDTA.anno
cm_raw <- read_tsv("T2T-Polish.fa.mod.EDTA.TE.fa.stat.redun.sum", skip = 1, show_col_types = FALSE)

# 转为 data.frame，并设定行名
cm_df <- as.data.frame(cm_raw)
rownames(cm_df) <- cm_df[[1]]
cm_df <- cm_df[, -1]  # 去掉第一列（行名列）

# 转为矩阵
cm <- as.matrix(cm_df)
# 去掉 Misclas_rate 列
cm2 <- cm[, -ncol(cm)]
# 计算每类真实类型总数
row_totals <- rowSums(cm2)
# 总数量
grand_total <- sum(cm2)
# 各类型占比
type_proportion <- row_totals / grand_total
# 每类正确预测数（对角线）
true_pos <- diag(cm2)
# 每类准确率
accuracy_per_type <- true_pos / row_totals
# 输出数据框
result <- data.frame(
  Type = rownames(cm2),
  Total = row_totals,
  Proportion = round(type_proportion, 4),
  True_Pos = true_pos,
  Accuracy = round(accuracy_per_type, 4)
)
print(result)

library(dplyr)
library(ggplot2)

library(scales)      # alpha()

  )

p_acc <- ggplot(result,

  scale_fill_manual(values = group_colors, guide = "none") +
