#### **（2）统计六大数据库注释的基因数量**
```bash
library(tidyverse)

file <- "DYGY.emapper.all.annotations"

## ── 1. 找到真正的表头行（第一个以 "#query" 开头的行） ─────────────────────────
header_line <- grep("^#query", readLines(file), value = FALSE)[1]  # 返回行号

## ── 2. 用 read_tsv 读入：skip 掉 header_line-1 行；col_names = TRUE ───────────
anno <- read_tsv(
  file,
  skip = header_line - 1,   # 把之前的 meta 全部跳过
  comment = "",             # 不能再把 # 当注释
  col_names = TRUE,
  show_col_types = FALSE
)

## ── 3. 为避免 # 号残留，去掉列名里的 # ────────────────────────────────────────
colnames(anno) <- sub("^#", "", colnames(anno))

## ── 4. 标记六大数据库是否有注释 ───────────────────────────────────────────────
anno <- anno %>%
  mutate(
    NR         = !is.na(Description)    & Description    != "-",
    SwissProt  = !is.na(Preferred_name) & Preferred_name != "-",
    GO         = !is.na(GOs)            & GOs            != "-",
    KEGG       = !is.na(KEGG_ko)        & KEGG_ko        != "-",
    Pfam       = !is.na(PFAMs)          & PFAMs          != "-",
    InterPro   = !is.na(PFAMs)          & PFAMs          != "-"   # eggNOG 将 InterPro/Pfam 一起放在 PFAMs
  )

## ── 5. 统计每个数据库的基因数和比例 ───────────────────────────────────────────
total_genes <- nrow(anno)

summary_tbl <- anno %>%
  summarise(across(c(NR, SwissProt, GO, KEGG, Pfam, InterPro), sum)) %>%
  pivot_longer(everything(), names_to = "Database", values_to = "Gene_Count") %>%
  mutate(Percent = round(Gene_Count / total_genes * 100, 2))

print(summary_tbl)
```

```bash
#🧬 Step 3. Map hit proteins to OGs (Orthologous Groups)
# -*- coding: utf-8 -*-
import pandas as pd
import json
import glob
import os

print("Script started.")
print("Current working directory:", os.getcwd())

# =========================
# Load functional annotation JSON (line-delimited)
# =========================
func_file = "./data/e6.func_profiles.json"
func = {}
print(f"Loading functional annotations from {func_file} ...")
try:
    with open(func_file, encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            entry = json.loads(line)
            og = entry["n"]
            fprof = entry.get("fprof", {})
            func[og] = {
                "GO": [i["n"] for i in fprof.get("GOslim", [])],
                "KEGG": [i["n"] for i in fprof.get("kegg_ko", [])],
                "COG": ";".join([i["n"] for i in fprof.get("pfam_arch", [])]),
                "Description": ";".join([i["n"] for i in fprof.get("best_name", [])])
            }
    print("Functional annotation loaded. Number of OGs:", len(func))
except Exception as e:
    print("Error loading functional annotation:", e)
    func = {}

def get_annotation(og):
    og = str(og)
    if og in func:
        entry = func[og]
        return pd.Series({
            "GO": ";".join(entry.get("GO", [])),
            "KEGG": ";".join(entry.get("KEGG", [])),
            "COG": entry.get("COG", ""),
            "Description": entry.get("Description", "")
        })
    else:
        return pd.Series({"GO": "", "KEGG": "", "COG": "", "Description": ""})

# =========================
# Load protein -> OG mapping
# =========================
seq2ogs_file = "./data/e6.seq2ogs.tsv"
print(f"Loading protein -> OG mapping from {seq2ogs_file} ...")
try:
    seq2ogs = pd.read_csv(seq2ogs_file, sep="\t", header=None, names=["protein", "OGs"])
    seq2ogs["OG_list"] = seq2ogs["OGs"].str.split(",")
    seq2ogs["OG"] = seq2ogs["OG_list"].apply(lambda x: x[0] if x else None)
    protein2og = dict(zip(seq2ogs["protein"], seq2ogs["OG"]))
    print("Protein->OG mapping loaded. Number of proteins:", len(protein2og))
except Exception as e:
    print("Error loading seq2ogs.tsv:", e)
    protein2og = {}

# =========================
# Process all diamond files
# =========================
diamond_files = glob.glob("*.diamond.tsv")
print("Found diamond files:", diamond_files)

if not diamond_files:
    print("No diamond files found in current directory. Exiting.")
    exit()

for diamond_file in diamond_files:
    species = os.path.basename(diamond_file).replace(".diamond.tsv", "")
    print("\nProcessing species:", species)

        # Correct column names for Diamond output

            sep="\t",

            names=["query", "hit_protein", "pident", "length", "evalue", "bitscore"]
        )
        print("Diamond file rows:", len(diamond))

    except Exception as e:
        print(f"Error reading {diamond_file}:", e)

    diamond["OG"] = diamond["hit_protein"].map(protein2og)
    missing_count = diamond["OG"].isna().sum()
    if missing_count > 0:

    # Map OG to functional annotation
    annotations = diamond["OG"].apply(get_annotation)
```
