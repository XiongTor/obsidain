### 19.大叶桂樱叶绿体和线粒体基因组组装和注释以及可视化

```bash
#路径/data/xueqin/Project/Prunus/Prunus_Geome_analyses/Prunus_zippeliana_genome/T2T_final/1delete_chro_mit_2026.10.12/01_organelle_filter
#!/usr/bin/env bash
# ================================================================
# Prunus zippeliana — 细胞器来源 contig 筛查与 Hi-C 挂载追踪
# 文件：readme.txt
# 目的：从 hifiasm primary assembly 中筛查叶绿体、线粒体候选 contigs，
#       再利用 JBAT AGP 检查候选 contigs 是否参与 Hi-C 挂载。
# 说明：所有解释均用 # 注释；运行至最后不会自动删除任何 contig。
# 运行位置：T2T_final/1delete_chro_mit_2026.10.12/
# 依赖：awk、minimap2、seqkit、python3、coreutils
# 注意：根据实际目录确认输入文件存在；本流程会覆盖同名中间结果。
# ================================================================

set -euo pipefail

# STEP 0：回顾 hifiasm 内部去冗余记录和组装文件（在 T2T_final 目录）
# 已有日志显示 [M::purge_dups]，说明 hifiasm 执行过内部去冗余相关处理；
# 不能仅据此认定不存在残留冗余，异源四倍体更不能直接删除同源亚基因组。
T2T_DIR="/data/xueqin/Project/Prunus/Prunus_Geome_analyses/Prunus_zippeliana_genome/T2T_final"
grep -iE 'purge|dup|coverage|hom-cov|haplotig' "$T2T_DIR/out.log" | head -100 || true
ls -lh "$T2T_DIR"/Prunus_zippeliana*.gfa

# STEP 1：建立工作目录和输入软链接
WORK="$T2T_DIR/1delete_chro_mit_2026.10.12"
mkdir -p "$WORK/01_organelle_filter"
cd "$WORK"
# 避免重复创建软链接导致报错；链接不修改源文件。
if [[ ! -e Prunus_zippeliana.bp.p_ctg.fa ]]; then
    ln -s ../Prunus_zippeliana.bp.p_ctg.fa ./
fi
# 此处应已有 himt_chloroplast.gfa 和 himt_mitochondrial.fasta。
for f in himt_chloroplast.gfa himt_mitochondrial.fasta Prunus_zippeliana.bp.p_ctg.fa; do
    [[ -s "$f" ]] || { echo "ERROR: missing $WORK/$f" >&2; exit 1; }
done
cd 01_organelle_filter

# STEP 2：准备细胞器参考序列
# 叶绿体 GFA：提取 S 行的 segment 序列；不等于重构了完整叶绿体环状基因组。
awk '$1=="S" && $3!="*" {print ">"$2"\n"$3}' ../himt_chloroplast.gfa > chloroplast.fa
cp ../himt_mitochondrial.fasta mitochondria.fa
seqkit stats -a chloroplast.fa mitochondria.fa
# 可根据当前环境使用 conda activate base；本脚本不自动切换 conda 环境。

# STEP 3：核基因组 contigs 对两类细胞器参考的比对
# Query 为 hifiasm primary contigs；PAF 第1、2、3、4列为 query 名称、长度及区间。
# Reference 为细胞器序列；asm5 适合高度相似序列，-c 输出 CIGAR 信息。
minimap2 -x asm5 -c -t 16 chloroplast.fa ../Prunus_zippeliana.bp.p_ctg.fa > chloroplast_vs_contigs.paf
minimap2 -x asm5 -c -t 16 mitochondria.fa ../Prunus_zippeliana.bp.p_ctg.fa > mitochondria_vs_contigs.paf

# STEP 4：按单条 PAF 记录筛选高覆盖候选（初筛，不是删除标准）
# 阈值：单条 query 覆盖率 >= 0.80，近似一致性(matches/block length) >= 0.95。
# 多个分段覆盖同一 contig 时此方法可能漏检，因此线粒体还要做区间合并。
awk 'BEGIN{OFS="\t"} {cov=($4-$3)/$2; identity=($11>0?$10/$11:0); if(cov>=0.8 && identity>=0.95) print $1,$2,$6,$3,$4,cov,identity,$12}' chloroplast_vs_contigs.paf > chloroplast_candidates.tsv
awk 'BEGIN{OFS="\t"} {cov=($4-$3)/$2; identity=($11>0?$10/$11:0); if(cov>=0.8 && identity>=0.95) print $1,$2,$6,$3,$4,cov,identity,$12}' mitochondria_vs_contigs.paf > mitochondria_candidates.tsv
column -t chloroplast_candidates.tsv | head -30 || true
column -t mitochondria_candidates.tsv | head -30 || true

# STEP 5：叶绿体候选的数量、总长度与序列提取
# 本次实际分析得到36条叶绿体候选；重新运行后以新输出为准。
cut -f1 chloroplast_candidates.tsv | sort -u > chloroplast_candidate_ids.txt
wc -l chloroplast_candidate_ids.txt
awk '!seen[$1]++ {sum+=$2; n++} END {print "Candidate contigs:",n+0; printf "Total length: %.3f Mb\n",sum/1000000}' chloroplast_candidates.tsv
# edge_451 是 HiMT 叶绿体 GFA 中一个 87,031 bp segment，而非叶绿体完整基因组。
seqkit grep -p edge_451 chloroplast.fa | seqkit stats -a
seqkit stats -a chloroplast.fa
seqkit grep -f chloroplast_candidate_ids.txt ../Prunus_zippeliana.bp.p_ctg.fa > chloroplast_candidate_contigs.fa
seqkit stats -a chloroplast_candidate_contigs.fa

# STEP 6：线粒体局部匹配检查
# 本次线粒体 PAF 共118行；单条覆盖率>=80%的列表为空，不能据此排除线粒体。
wc -l mitochondria_vs_contigs.paf
# len>=1000 bp、近似一致性>=90%，用于查看线粒体局部比对。
awk 'BEGIN{OFS="\t"} {len=$4-$3; identity=($11>0?$10/$11:0); if(len>=1000 && identity>=0.90) print $1,$2,$6,$3,$4,len,identity,$12}' mitochondria_vs_contigs.paf > mitochondria_hits_1kb.tsv
awk 'BEGIN{OFS="\t"} {len=$4-$3; cov=len/$2; identity=($11>0?$10/$11:0); if(len>=1000 && identity>=0.90) print $1,$2,$6,len,cov,identity,$12}' mitochondria_vs_contigs.paf | sort -k4,4nr > mitochondria_hits_review.tsv
column -t mitochondria_hits_review.tsv | head -40 || true

# STEP 7：创建脚本，计算同一 contig 上高一致性线粒体区间的合并覆盖率
# 使用 query 0-based 半开区间，合并重叠片段后计算非重复覆盖长度。
# 这是候选鉴定指标，仍需检查核 NUMT、测序深度及染色体连接证据。
cat > calculate_mito_coverage.py <<'PY'
#!/usr/bin/env python3
from collections import defaultdict

paf = "mitochondria_vs_contigs.paf"
out = "mitochondria_contig_coverage.tsv"
regions = defaultdict(list)
lengths = {}

with open(paf) as f:
    for line in f:
        a = line.rstrip("\n").split("\t")
        if len(a) < 12:
            continue
        contig, qlen = a[0], int(a[1])
        start, end = int(a[2]), int(a[3])
        matches, block = int(a[9]), int(a[10])
        if block == 0 or matches / block < 0.95:
            continue
        lengths[contig] = qlen
        regions[contig].append((start, end))

results = []
for contig, intervals in regions.items():
    intervals.sort()
    merged = []
    for start, end in intervals:
        if not merged or start > merged[-1][1]:
            merged.append([start, end])
        else:
            merged[-1][1] = max(merged[-1][1], end)
    covered = sum(end - start for start, end in merged)
    qlen = lengths[contig]
    results.append((contig, qlen, covered, covered / qlen, len(intervals)))

results.sort(key=lambda x: x[3], reverse=True)
with open(out, "w") as f:
    f.write("Contig\tLength\tCovered_bp\tCoverage\tHits\n")
    for contig, qlen, covered, cov, hits in results:
        f.write(f"{contig}\t{qlen}\t{covered}\t{cov:.4f}\t{hits}\n")
print("Output:", out)
print("Candidate contigs:", len(results))
PY
python calculate_mito_coverage.py
column -t mitochondria_contig_coverage.tsv | head -40 || true

# STEP 8：线粒体高覆盖候选提取及与叶绿体候选交集检查
# 本次运行10条 contigs 的线粒体参考累计覆盖率均超过99%。
# 使用 >=0.95 作为候选标准，不作为自动删除标准。
awk 'NR>1 && $4>=0.95 {print $1}' mitochondria_contig_coverage.tsv | sort -u > mitochondria_candidate_ids.txt
wc -l mitochondria_candidate_ids.txt
cat mitochondria_candidate_ids.txt
seqkit grep -f mitochondria_candidate_ids.txt ../Prunus_zippeliana.bp.p_ctg.fa > mitochondria_candidate_contigs.fa
seqkit stats -a mitochondria_candidate_contigs.fa
# 两组候选如有交集，可能涉及线粒体中的叶绿体来源片段，需进一步核对。
comm -12 <(sort -u chloroplast_candidate_ids.txt) <(sort -u mitochondria_candidate_ids.txt) > shared_organelle_candidate_ids.txt
cat shared_organelle_candidate_ids.txt

# STEP 9：利用 Hi-C 人工校正后的 AGP 追踪候选 contigs
# AGP 第5列为组件类型；W/D 表示有序列的组件，第6列通常为原始 contig ID。
# 注意：AGP 可能包含名称调整、拆分或不同组件格式，查不到不等于未挂载。
AGP_SRC="/data/xueqin/Project/Prunus/Prunus_Geome_analyses/Prunus_zippeliana_genome/Haphic_t2t/result/04.build/out_JBAT.FINAL.agp"
[[ -s "$AGP_SRC" ]] || { echo "ERROR: AGP not found: $AGP_SRC" >&2; exit 1; }
ln -sfn "$AGP_SRC" out_JBAT.FINAL.agp
head -n 10 out_JBAT.FINAL.agp
awk '!/^#/ {count[$5]++} END {for (x in count) print x,count[x]}' out_JBAT.FINAL.agp
awk 'NR==FNR {ids[$1]=1; next} ($5=="W" || $5=="D") && ($6 in ids) {print}' mitochondria_candidate_ids.txt out_JBAT.FINAL.agp > mitochondria_in_HiC.tsv
awk 'NR==FNR {ids[$1]=1; next} ($5=="W" || $5=="D") && ($6 in ids) {print}' chloroplast_candidate_ids.txt out_JBAT.FINAL.agp > chloroplast_in_HiC.tsv
cat mitochondria_in_HiC.tsv
cat chloroplast_in_HiC.tsv
# 统计 AGP 中能匹配到的候选 contig 个数（不是组件行数）。
echo "Mitochondrial candidates in Hi-C AGP:"
awk '{print $6}' mitochondria_in_HiC.tsv | sort -u | grep -c . || true
echo "Chloroplast candidates in Hi-C AGP:"
awk '{print $6}' chloroplast_in_HiC.tsv | sort -u | grep -c . || true

# STEP 10：结果解释与后续工作（仅记录，不执行删除）
# 当前记录：叶绿体单条比对候选36条；线粒体合并覆盖率>=95%候选10条。
# 原始 hifiasm 已出现 purge_dups 内部处理日志，但细胞器去除需另行判断。
# 继续核对：Hi-C AGP 中候选是否属于16条核染色体、HiFi/ONT 深度、
#           reads 跨越连接位置、是否为核内真实 NUPT/NUMT。
# 若候选被错误挂载，还需追踪至 Gap filling/Polishing 后版本的准确坐标。
# 切勿直接使用上述候选列表自动删除整个 contig 或染色体内部片段。
# ================================================================
```
